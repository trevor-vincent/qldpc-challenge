"""Validate committed native-search witnesses and optionally replay one recorded call.

Use --records for the evidence index, --definition for the explicit Clifford
construction, and --out for a new report. --replay-key requires --replay-out;
its native return is saved before any validation. Bounds remain upper bounds.
"""

import argparse
import hashlib
import json
import sys
import time
from pathlib import Path

import numpy as np
from reproduce_pair_clifford_662_122 import ROOT, reconstruct

sys.path.insert(0, str(ROOT / "verify"))
import heuristic_distance as hd
from gf2 import logical_basis


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def map_pauli(v, pairs):
    n = len(v) // 2
    result = v.copy()
    for left, right in pairs:
        x1, x2, z1, z2 = (v[i] for i in [left, right, n + left, n + right])
        result[left], result[right] = x2 ^ z2, x1 ^ z1
        result[n + left], result[n + right] = x1 ^ x2 ^ z1, x1 ^ x2 ^ z2
    return result


def make_problems(definition):
    hx, hz, s = reconstruct(definition)
    n = 2 * definition["m"]
    a, b = s[:, :n], s[:, n:]
    full_f = np.c_[b, a]
    logicals = logical_basis(full_f, s)
    full_lf = np.c_[logicals[:, n:], logicals[:, :n]]
    lx, lz = logical_basis(hz, hx), logical_basis(hx, hz)
    pairs = definition["pairs"]
    untouched = sorted(set(range(n)) - {q for pair in pairs for q in pair})
    restriction = np.zeros((len(untouched), 2 * n), dtype=np.int8)
    restriction[np.arange(len(untouched)), untouched] = 1
    restriction[np.arange(len(untouched)), np.array(untouched) + n] = 1
    problems = {
        "parent-Y": {"F": np.vstack([hx, hz]), "Lf": np.vstack([lx, lz]), "weights": [1] * n, "pauli_c": 0},
        "full-Pauli": {"F": full_f, "Lf": full_lf, "weights": [1] * n, "pauli_c": n},
        "extended-Y": {"F": np.vstack([full_f, restriction]), "Lf": full_lf, "weights": [1] * n, "pauli_c": n},
    }
    t, c = len(pairs), len(pairs) + len(untouched)
    positions = np.empty(n, dtype=int)
    for i, (left, right) in enumerate(pairs):
        positions[left], positions[right] = i, c + i
    for i, q in enumerate(untouched):
        positions[q] = t + i
    embedding = np.zeros((n, 2 * c), dtype=np.int8)
    embedding[np.arange(n), positions] = 1
    fixed = np.zeros((len(untouched), 2 * c), dtype=np.int8)
    fixed[np.arange(len(untouched)), c + t + np.arange(len(untouched))] = 1
    for side, kernel, opposite in [("X", hz, lz), ("Z", hx, lx)]:
        problems["pure-" + side] = {
            "F": np.vstack([kernel @ embedding % 2, fixed]),
            "Lf": opposite @ embedding % 2,
            "weights": [2] * t + [1] * len(untouched),
            "pauli_c": c,
            "positions": positions,
        }
    return problems, a, b


def validate_raw(raw, kind, problem, a, b, pairs):
    n = a.shape[1]
    support_key = "support" if kind == "parent-Y" and "support" in raw else "support_flat"
    v = np.zeros(problem["F"].shape[1], dtype=np.int8)
    v[raw[support_key]] = 1
    assert not (problem["F"] @ v % 2).any() and (problem["Lf"] @ v % 2).any()
    c = problem["pauli_c"]
    measured = int((v[:c] | v[c:]) @ np.asarray(problem["weights"])) if c else int(v.sum())
    assert measured == raw["weight"]
    if kind == "parent-Y":
        logical = map_pauli(np.r_[v, v], pairs)
    elif kind.startswith("pure-"):
        parent = v[problem["positions"]]
        logical = np.zeros(2 * n, dtype=np.int8)
        logical[slice(0, n) if kind == "pure-X" else slice(n, 2 * n)] = parent
        logical = map_pauli(logical, pairs)
    else:
        logical = v
    assert hd.valid_pauli_logical(logical, a, b)
    assert int((logical[:n] | logical[n:]).sum()) == raw["weight"]
    return logical


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--definition", required=True, type=Path)
    parser.add_argument("--records", required=True, type=Path)
    parser.add_argument("--out", required=True, type=Path)
    parser.add_argument("--replay-key")
    parser.add_argument("--replay-out", type=Path)
    args = parser.parse_args()
    assert not args.out.exists(), "Choose a new report path."
    if args.replay_key:
        assert args.replay_out is not None and not args.replay_out.exists()
        args.replay_out.mkdir(parents=True)
    definition = json.loads(args.definition.read_text())
    index = json.loads(args.records.read_text())
    problems, a, b = make_problems(definition)
    report = {
        "status": "running",
        "scope": "Native witness validation; all distances are upper bounds.",
        "definition_sha256": digest(args.definition),
        "index_sha256": digest(args.records),
        "script_sha256": digest(Path(__file__)),
        "records": [],
    }
    args.out.parent.mkdir(parents=True, exist_ok=True)

    def persist():
        args.out.write_text(json.dumps(report, indent=2) + "\n")

    persist()
    replay_found = False
    for row in index["records"]:
        raw_path = args.records.parent / row["raw"]
        assert digest(raw_path) == row["raw_sha256"]
        raw = json.loads(raw_path.read_text())
        kind = row["kind"]
        problem = problems[kind]
        for key in ["F", "Lf"]:
            assert hashlib.sha256(problem[key].astype(np.int8).tobytes()).hexdigest() == row[key + "_sha256"]
        assert problem["pauli_c"] == row["pauli_c"]
        logical = validate_raw(raw, kind, problem, a, b, definition["pairs"])
        record = {
            "id": row["id"],
            "kind": kind,
            "weight": raw["weight"],
            "validated": True,
            "mapped_witness": hd.pauli_witness(logical, a.shape[1]),
        }
        if row["id"] == args.replay_key:
            import gf2_fast

            assert digest(ROOT / "verify/gf2_fast.cpp") == index["native_source_sha256"]
            parameters = {key: raw[key] for key in ["seed", "pair_depth", "exhaustive_dim", "threads"]}
            parameters["trials"] = raw["trials"] if "trials" in raw else raw["trials_requested"]
            start = time.monotonic()
            weight, support = gf2_fast.fold_rand_witness(
                problem["F"], problem["Lf"], problem["weights"], pauli_c=problem["pauli_c"], **parameters
            )
            support_key = "support" if kind == "parent-Y" and "support" in raw else "support_flat"
            returned = parameters | {
                "pauli_c": problem["pauli_c"],
                "weight": int(weight),
                support_key: list(map(int, support)),
                "seconds": time.monotonic() - start,
            }
            path = args.replay_out / (row["id"] + "-raw.json")
            path.write_text(json.dumps(returned, indent=2) + "\n")
            replayed_logical = validate_raw(returned, kind, problem, a, b, definition["pairs"])
            if index.get("candidate"):
                import copy
                from qldpc_verify import verify
                from submit import save_submission

                current = json.loads((ROOT / index["candidate"]).read_text())
                if int(weight) < current["distance"]["d"]:
                    revised = copy.deepcopy(current)
                    revised["distance"] = {"d": int(weight), "P": {
                        "value": int(weight), "confidence": "upper_bound",
                        "witness": hd.pauli_witness(replayed_logical, a.shape[1])}}
                    revised["name"] = f"[[{current['n']},{current['k']},{int(weight)}]] repaired pair-Clifford image"
                    corrected = args.replay_out / f"{current['n']}-{current['k']}-{int(weight)}.json"
                    assert not save_submission(revised, str(corrected))
                    checked = verify(revised)
                    corrected.with_suffix('.structural.json').write_text(json.dumps(checked, indent=2) + "\n")
                    assert checked["ok"], checked
                    record["corrected_candidate"] = str(corrected)
            record["replay"] = {
                "raw": str(path),
                "native_sha256": digest(Path(gf2_fast.__file__)),
                "weight": int(weight),
                "exact_weight_and_support_match": returned["weight"] == raw["weight"]
                and returned[support_key] == raw[support_key],
            }
            replay_found = True
        report["records"].append(record)
        persist()
    assert not args.replay_key or replay_found, "Unknown replay key."
    report["status"] = "completed"
    persist()
    print(
        json.dumps({"status": "completed", "validated_records": len(report["records"]), "replayed": args.replay_key}),
        flush=True,
    )


if __name__ == "__main__":
    main()
